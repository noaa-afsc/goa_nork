SELECT
  "SURVEY_DEFINITION_ID" AS "survey_definition_id",
  "YEAR" AS "year",
  "AREA_ID" AS "area_id",
  "SPECIES_CODE" AS "species_code",
  "LENGTH_MM" AS "length_mm",
  "SEX" AS "sex",
  "POPULATION_COUNT" AS "population_count",
  "AKFIN_LOAD_DATE" AS "akfin_load_date"
FROM (gap_products.akfin_sizecomp) "q01"
WHERE
  ("SURVEY_DEFINITION_ID" IN (47.0)) AND
  ("AREA_ID" IN (99903.0)) AND
  ("SPECIES_CODE" IN (30420.0))
